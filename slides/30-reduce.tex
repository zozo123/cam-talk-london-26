\begin{frame}{Fork $\ne$ independence: the cold-liar problem}
\Large
A child should not return a naked score.

\vspace{0.4em}
\begin{center}
\Large
$r_k=(\hat\theta_k,\; J_k,\; n_k,\; \mathcal{E}_k,\; \mathcal{L}_k,\; m_k)$
\end{center}

\vspace{0.5em}
\normalsize
\begin{itemize}
  \item $\hat\theta_k$: estimate or candidate result
  \item $J_k, n_k$: precision / sample size; in the Gaussian reduce, $\beta_k \equiv n_k$
  \item $\mathcal{E}_k, \mathcal{L}_k$: evidence IDs and fork lineage
  \item $m_k$: method / model / harness identity
\end{itemize}

\begin{block}{Cold liar}
If a worker can forge high precision, it hijacks the pool. Isolation does not authenticate $n_k$.
\end{block}

\source{Evidence-Aware MapReduce / Boltzmann MapReduce: Eliaz, arXiv:2607.09689.}
\end{frame}

\begin{frame}{Annealing is schedule language, not mysticism}
\normalsize
\begin{tabular}{p{0.24\textwidth}p{0.64\textwidth}}

\textbf{Cousin} & \textbf{Factory translation} \\
\midrule

Metropolis-Hastings & propose $\Delta$ in child; keep = accept; revert = burn \\
SGD / Langevin & noisy local ascent on one path; step/noise policy outside worker \\
RL temperature & entropy / KL exploration is an Epoch policy; checkpoint is the tip \\
\bottomrule

\vspace{0.8em}
\begin{block}{Thermostat rule}
The worker may propose. The controller owns $T$, $\beta$, energy, acceptance, and the schedule digest.
\end{block}

\vspace{0.4em}
\centering \large Factory = driven/open system; promote = absorbing sink, not equilibrium proof.
\end{frame}
