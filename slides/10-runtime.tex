\begin{frame}{What actually has to fork?}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{State}
\begin{itemize}
  \item \term{filesystem}: named workspace snapshot
  \item \term{memory + processes}: restore the running machine
  \item \term{reproducibility}: identify $S_0$ by digest
\end{itemize}
\end{block}

\begin{block}{Speed}
\begin{itemize}
  \item \term{CoW cloning}: pay on divergence, not on rebuild
  \item \term{warm build/test}: share heat beside the trust boundary
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Authority}
\begin{itemize}
  \item \term{network}: explicit egress, not ambient access
  \item \term{credentials}: remint / lease, never blindly copy
  \item \term{oracle}: tests/reward stay outside worker write-set
\end{itemize}
\end{block}

\begin{block}{Evidence}
\begin{itemize}
  \item logs, digests, lineage outlive the child
  \item crash/retry must not change who owns the decision
\end{itemize}
\end{block}
\end{columns}

\end{frame}
