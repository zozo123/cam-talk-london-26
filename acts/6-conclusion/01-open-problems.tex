\begin{frame}[t]{\ev{\tagP}Two concrete open problems}
\vspace{.1cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{1\enspace Dependence model}
{\small Use recorded ancestry, fixtures and feedback to predict outcome dependence. Validate the calibration on unseen task families.\par}
\vspace{.15cm}
{\small\color{Teal}\textbf{Useful output.} An estimator, a conservative bound or a refusal rule that improves held-out calibration.\par}
\end{block}
\column{.48\textwidth}
\begin{block}{2\enspace Authority after restore}
{\small Reissue child authority with a generation, an expiry and a budget. Reject stale publication across restore and failure.\par}
\vspace{.15cm}
{\small\color{Teal}\textbf{Useful output.} A checked protocol covering clone, expiry, stale-generation and uncertain-effect cases.\par}
\end{block}
\end{columns}
\vspace{.35cm}
{\small\color{Muted}\textbf{Extensions.} Warm-state placement, side channels, long human waits and effects on remote services.\par}
