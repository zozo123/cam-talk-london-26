\begin{frame}[t]{The fork contract, in full}
\vspace{0pt}
{\small\textbf{Fork contract, written before the fork}\par\vspace{2pt}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\begin{enumerate}\setlength{\itemsep}{0pt}
\item fork from an immutable, content-addressed parent
\item each branch: own lease, TTL, budget, provenance
\item no publishing credentials in a coding cell
\item record parent checkpoint, branch ID, evidence IDs
\end{enumerate}
\column{.49\textwidth}
\begin{enumerate}\setcounter{enumi}{4}\setlength{\itemsep}{0pt}
\item {\color{Rust}\bfseries never count siblings that reuse the same evidence as independent agreement}
\item merge deterministically; conflicts are evidence
\item full verification and review on the merged result
\item tear down every child, on any exit \hfill{\color{Muted}our second run}
\end{enumerate}
\end{columns}\par}
