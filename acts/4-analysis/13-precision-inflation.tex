\begin{frame}[t]{\ev{\tagB\,\tagP}Claimed precision can be inflated}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\term{One shard inflates its precision}\\
{\small A 2,000-point shard returns a distant estimate and inflates its reported precision a further $50\times$.\par}
\vspace{.2cm}
{\small
\begin{ticks}
\item Claimed precision raises the shard's weight.
\item Sample count alone cannot detect inflated information per point.
\item The stress heuristic is not a Byzantine guarantee.
\end{ticks}\par}
