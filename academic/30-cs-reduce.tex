\begin{frame}{Reduce is not majority vote}
\small
\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
A worker should not return only:
\begin{center}
\Large \warn{score = 0.94}
\end{center}
It should return a structured record:
\begin{center}
\normalsize result + precision + evidence IDs + lineage + method
\end{center}
\column{0.48\textwidth}
\begin{block}{Why?}
\begin{itemize}
  \item two siblings may share the same bad root
  \item one worker may claim absurd confidence
  \item tests may be changed mid-epoch
  \item evidence may be missing or stale
\end{itemize}
\end{block}
\end{columns}
\vspace{0.6em}
\centering\large When the reducer cannot tell, \hilite{abstain} is a valid result.
\end{frame}

\begin{frame}{Four clones do not make four witnesses}
\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
\Large
\begin{center}
$S_0 \rightarrow \{w_1,w_2,w_3,w_4\}$
\end{center}
\normalsize They may share:
\begin{itemize}
  \item the same prompt and model
  \item the same cached artifact
  \item the same hidden bad fixture
  \item the same ancestor state
\end{itemize}
\column{0.48\textwidth}
\begin{block}{Cold liar}
If a worker can claim absurd confidence, it can dominate a reducer even when it is wrong.
\end{block}
\vspace{0.6em}
\begin{block}{Correlation}
Execution isolation is not statistical independence.
\end{block}
\end{columns}
\vspace{0.5em}
\centering\Large \hilite{Fork makes search cheap. It does not manufacture evidence.}
\source{Evidence-Aware / Boltzmann MapReduce: Eliaz, arXiv:2607.09689.}
\end{frame}
