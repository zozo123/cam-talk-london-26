% 4
\begin{frame}{Why I ended up here}
\begin{columns}[T,onlytextwidth]
\column{0.30\textwidth}
\textbf{Computational science}\par
\normalsize An experiment is not reproducible unless the state, inputs, machine, and provenance are reproducible.

\column{0.30\textwidth}
\textbf{Production ML + builds}\par
\normalsize Rebuilding the world is expensive. Reuse warm state whenever it is safe.

\column{0.30\textwidth}
\textbf{Coding agents}\par
\normalsize The developer becomes a program that edits files, runs shells, calls networks, retries, and forks hypotheses.
\end{columns}

\vspace{0.75em}
\begin{block}{The collision}
\centering
\large The same \textbf{state} I wanted to reuse for speed became the state I had to isolate for trust.
\end{block}
\end{frame}

% 5
\begin{frame}{Glossary: terms we will use carefully}
\small
\begin{tabular}{p{0.22\textwidth}p{0.68\textwidth}}
\toprule
\textbf{Term} & \textbf{Meaning in this talk} \\
\midrule
Agent & A program that plans, edits, runs tools, observes results, and tries again. \\
Sandbox & An execution boundary around files, processes, network, and capabilities. \\
Snapshot & A named starting state $S_0$ that can be restored or forked. \\
Fork & Create a disposable child world from $S_0$. \\
Rollout & One attempted trajectory: actions, observations, artifacts, logs. \\
Verifier / oracle & The evaluator: tests, reward, hidden checks, human or scientific measurement. \\
Authority & The power to merge, release, publish, spend robot time, or update a checkpoint. \\
Evidence & Durable proof of what happened: logs, digests, lineage, measurements. \\
\bottomrule
\end{tabular}
\end{frame}
