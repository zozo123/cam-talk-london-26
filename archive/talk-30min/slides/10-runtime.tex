\begin{frame}{There are two ways to make futures cheap}
\begin{columns}[T,onlytextwidth]
\column{0.47\textwidth}
\begin{block}{Learn a world}
\centering
\large Dreamer / Contrastive World Models
\end{block}
\normalsize
\begin{center}
$o_t \rightarrow z_t \rightarrow \widehat{s}_{t+1},\widehat{s}_{t+2},\ldots$
\end{center}
\begin{itemize}
  \item compact, learned latent state
  \item imagine many futures cheaply
  \item useful when interaction is slow, unsafe, or scarce
  \item pay approximation / learned-dynamics error
\end{itemize}
\small CWM's twist: stop spending capacity reconstructing irrelevant pixels; keep information predictive of the future.

\column{0.47\textwidth}
\begin{block}{Fork the world}
\centering
\large executable software environment
\end{block}
\normalsize
\begin{center}
$S_0 \rightarrow \{S_0^1,S_0^2,\ldots,S_0^N\}$
\end{center}
\begin{itemize}
  \item explicit, executable state
  \item branch real repo / process futures
  \item useful when the world is software and reset is cheap
  \item pay compute / storage, not learned transition-model error
\end{itemize}
\small Exact only inside the state and capability boundary we actually captured.
\end{columns}

\vspace{0.45em}
\begin{block}{The systems inversion}
\centering\large World models make \textbf{imagination} cheap. Forkable sandboxes make \textbf{interaction} cheap.
\end{block}
\source{Dreamer 4: Hafner et al., arXiv:2509.24527. Contrastive World Models: Li, arXiv:2609.22175.}
\end{frame}

\begin{frame}{A fork should copy state, not authority}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Copy into the child}
\begin{itemize}
  \item filesystem + process / memory state
  \item pinned source, dependencies, toolchain
  \item enough state to continue the experiment
\end{itemize}
\end{block}

\begin{block}{Share carefully}
\begin{itemize}
  \item content-addressed build artifacts
  \item immutable heat / caches inside one trust domain
  \item CoW pages whose sharing policy is explicit
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Remint at the boundary}
\begin{itemize}
  \item API / cloud credentials
  \item network capabilities and identity
  \item randomness / epoch identity when required
\end{itemize}
\end{block}

\begin{block}{Keep outside the child}
\begin{itemize}
  \item hidden verifier + objective digest
  \item promotion key / merge authority
  \item durable evidence ledger
\end{itemize}
\end{block}
\end{columns}

\vspace{0.35em}
\begin{center}
\Large \hilite{Copy the machine, not the keys.}
\end{center}
\end{frame}
